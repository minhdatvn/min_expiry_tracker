class _AppLocalizationsDelegate
    extends LocalizationsDelegate<AppLocalizations> {
  const _AppLocalizationsDelegate();

  @override
  Future<AppLocalizations> load(Locale locale) {
    return SynchronousFuture<AppLocalizations>(lookupAppLocalizations(locale));
  }

  @override
  bool isSupported(Locale locale) =>
      <String>['en', 'vi'].contains(locale.languageCode);

  @override
  bool shouldReload(_AppLocalizationsDelegate old) => false;
}

AppLocalizations lookupAppLocalizations(Locale locale) {
  // Lookup logic when only language code is specified.
  switch (locale.languageCode) {
    case 'en':
      return AppLocalizationsEn();
    case 'vi':
      return AppLocalizationsVi();
  }

  throw FlutterError(
    'AppLocalizations.delegate failed to load unsupported locale "$locale". This is likely '
    'an issue with the localizations generation tool. Please file an issue '
    'on GitHub with a reproducible sample app and the gen-l10n configuration '
    'that was used.',
  );
}

import 'dart:convert';

// 1. Dịch Danh mục (giữ nguyên tên tự nhập nếu có)
String localizeCategory(AppLocalizations l10n, String category) {
  switch (category) {
    case 'All':
      return l10n.all;
    case 'Medicine':
      return l10n.catMedicine;
    case 'Cosmetics':
      return l10n.catCosmetics;
    case 'Food':
      return l10n.catFood;
    case 'Other':
      return l10n.catOther;
    default:
      return category;
  }
}

// 2. Dịch Chủ sở hữu gợi ý mặc định ('Me', 'Family'); giữ nguyên tên tự thêm (VD: 'Mẹ', 'Bé')
String localizeOwner(AppLocalizations l10n, String owner) {
  switch (owner) {
    case 'All':
      return l10n.allOwners;
    case 'Me':
      return l10n.ownerMe;
    case 'Family':
      return l10n.ownerFamily;
    case 'Unassigned':
      return l10n.unassigned;
    default:
      return owner;
  }
}

// 3. Dịch Vị trí gợi ý mặc định ('Fridge', 'Medicine Box', 'Bathroom'); giữ nguyên vị trí tự thêm
String localizeLocation(AppLocalizations l10n, String location) {
  switch (location) {
    case 'All':
      return l10n.allLocations;
    case 'Fridge':
      return l10n.locFridge;
    case 'Medicine Box':
      return l10n.locMedicineBox;
    case 'Bathroom':
      return l10n.locBathroom;
    case 'No Location':
      return l10n.noLocation;
    default:
      return location;
  }
}

// 4. Dịch nhóm hành động trên màn hình Log (ADD, UPDATE, NOTIFY, CALENDAR...)
String localizeLogAction(AppLocalizations l10n, String action) {
  switch (action) {
    case 'All':
      return l10n.all;
    case 'ADD':
      return l10n.logActionAdd;
    case 'UPDATE':
      return l10n.logActionUpdate;
    case 'DELETE':
      return l10n.logActionDelete;
    case 'NOTIFY':
      return l10n.logActionNotify;
    case 'CALENDAR':
      return l10n.logActionCalendar;
    case 'BACKUP':
      return l10n.logActionBackup;
    case 'ERROR':
      return l10n.logActionError;
    default:
      return action;
  }
}

// 5. Đóng gói thông điệp Log thành chuỗi JSON có cấu trúc để dịch được sang mọi ngôn ngữ
String encodeLogMessage(String key, [Map<String, dynamic>? params]) {
  return jsonEncode({
    'key': key,
    if (params != null) ...params,
  });
}

// 6. Giải mã chuỗi Log từ SQLite và hiển thị đúng ngôn ngữ hiện tại
String localizeLogMessage(AppLocalizations l10n, String rawMessage) {
  if (!rawMessage.startsWith('{')) {
    if (rawMessage == 'Sent test notification.') {
      return l10n.logTestNotifySent;
    }
    final addedMatch =
        RegExp(r'^Added "(.+)" \((.+)\)\.$').firstMatch(rawMessage);
    if (addedMatch != null) {
      return l10n.logItemAdded(
        addedMatch.group(1)!,
        localizeCategory(l10n, addedMatch.group(2)!),
      );
    }
    final updatedMatch = RegExp(r'^Updated "(.+)"\.$').firstMatch(rawMessage);
    if (updatedMatch != null) {
      return l10n.logItemUpdated(updatedMatch.group(1)!);
    }
    final deletedMatch = RegExp(r'^Deleted "(.+)"\.$').firstMatch(rawMessage);
    if (deletedMatch != null) {
      return l10n.logItemDeleted(deletedMatch.group(1)!);
    }
    final notifyMatch =
        RegExp(r'^Scheduled (\d+) reminder\(s\) for "(.+)"\.$')
            .firstMatch(rawMessage);
    if (notifyMatch != null) {
      return l10n.logRemindersScheduled(
        int.parse(notifyMatch.group(1)!),
        notifyMatch.group(2)!,
      );
    }
    return rawMessage;
  }

  try {
    final Map<String, dynamic> data = jsonDecode(rawMessage);
    final String key = data['key'] ?? '';

    // Tạo nhãn mốc thời gian song ngữ nếu log có lưu alertValue & alertUnit
    String formatAlertParam() {
      final int val = data['alertValue'] ?? 1;
      final String unit = data['alertUnit'] ?? 'days';
      return unit == 'hours'
          ? l10n.hoursBeforeLabel(val)
          : l10n.daysBeforeLabel(val);
    }

    switch (key) {
      case 'logItemAdded':
        return l10n.logItemAdded(
          data['name'] ?? '',
          localizeCategory(l10n, data['category'] ?? ''),
        );
      case 'logItemUpdated':
        return l10n.logItemUpdated(data['name'] ?? '');
      case 'logItemDeleted':
        return l10n.logItemDeleted(data['name'] ?? '');
      case 'logTestNotifySent':
        return l10n.logTestNotifySent;
      case 'logRemindersScheduled':
        return l10n.logRemindersScheduled(
          data['count'] ?? 0,
          data['name'] ?? '',
        );
      case 'logCalendarDenied':
        return l10n.logCalendarDenied;
      case 'logCalendarNoAccount':
        return l10n.logCalendarNoAccount;
      case 'logCalendarNoWritable':
        return l10n.logCalendarNoWritable;
      case 'logCalendarSkippedPast':
        return l10n.logCalendarSkippedPast(
          formatAlertParam(),
          data['name'] ?? '',
        );
      case 'logCalendarAdded':
        return l10n.logCalendarAdded(
          data['name'] ?? '',
          formatAlertParam(),
        );
      case 'logCalendarFailed':
        return l10n.logCalendarFailed(data['name'] ?? '');
      case 'logCalendarRemoved':
        return l10n.logCalendarRemoved(
          data['count'] ?? 0,
          data['name'] ?? '',
        );
      case 'logBackupExported':
        return l10n.logBackupExported(
          data['count'] ?? 0,
          data['fileName'] ?? '',
        );
      case 'logRestoreInvalid':
        return l10n.logRestoreInvalid;
      case 'logBackupRestored':
        return l10n.logBackupRestored(data['count'] ?? 0);
      default:
        return rawMessage;
    }
  } catch (_) {
    return rawMessage;
  }
}

// 7. Chuyển mốc cảnh báo sang nhãn song ngữ
String localizeAlertLabel(AppLocalizations l10n, AlertMilestone alert) {
  if (alert.unit == 'hours') {
    return l10n.hoursBeforeLabel(alert.value);
  }
  return l10n.daysBeforeLabel(alert.value);
}

// 8. Định dạng ngày/giờ theo ngôn ngữ
String formatDateByLocale(
  BuildContext context,
  DateTime date, {
  bool includeTime = false,
}) {
  final isVi = Localizations.localeOf(context).languageCode == 'vi';
  if (isVi) {
    return includeTime
        ? DateFormat('dd/MM, HH:mm').format(date)
        : DateFormat('dd/MM/yyyy').format(date);
  }
  return includeTime
      ? DateFormat('MMM dd, HH:mm').format(date)
      : DateFormat('MMM dd, yyyy').format(date);
}

class AlertMilestone {
  final int value;               // Con số (VD: 1, 3, 6, 30)
  final String unit;             // 'days' hoặc 'hours'
  final bool addToCalendar;      // True nếu người dùng tick ô "Thêm vào Lịch"
  final String? calendarEventId; // Mã sự kiện Android trả về để sau này tự xóa/sửa
  final String? calendarId;      // MỚI: Lưu luôn ID sổ Lịch đã ghi sự kiện để xóa đúng chỗ

  const AlertMilestone({
    required this.value,
    this.unit = 'days',
    this.addToCalendar = false,
    this.calendarEventId,
    this.calendarId,
  });

  // Hiển thị chữ mặc định cho tiêu đề sự kiện Lịch
  String get label {
    if (unit == 'hours') {
      return value == 1 ? '1 hour before' : '$value hours before';
    }
    return value == 1 ? '1 day before' : '$value days before';
  }

  // Quy đổi ra giờ để sắp xếp các mốc từ xa đến gần
  int get totalHours => unit == 'hours' ? value : value * 24;

  // Tạo bản sao có thay đổi một vài trường (cờ clearCalendarEventId xóa sạch cả ID sự kiện lẫn ID sổ Lịch)
  AlertMilestone copyWith({
    int? value,
    String? unit,
    bool? addToCalendar,
    String? calendarEventId,
    String? calendarId,
    bool clearCalendarEventId = false,
  }) {
    return AlertMilestone(
      value: value ?? this.value,
      unit: unit ?? this.unit,
      addToCalendar: addToCalendar ?? this.addToCalendar,
      calendarEventId: clearCalendarEventId
          ? null
          : (calendarEventId ?? this.calendarEventId),
      calendarId: clearCalendarEventId
          ? null
          : (calendarId ?? this.calendarId),
    );
  }

  Map<String, dynamic> toMap() {
    return {
      'value': value,
      'unit': unit,
      'addToCalendar': addToCalendar,
      'calendarEventId': calendarEventId,
      'calendarId': calendarId,
    };
  }

  factory AlertMilestone.fromMap(Map<String, dynamic> map) {
    return AlertMilestone(
      value: (map['value'] as num?)?.toInt() ?? 1,
      unit: (map['unit'] ?? 'days').toString(),
      addToCalendar: map['addToCalendar'] == true,
      calendarEventId: map['calendarEventId'] as String?,
      calendarId: map['calendarId'] as String?,
    );
  }
}

class HouseholdItem {
  // --- THÔNG TIN BẮT BUỘC ---
  final String name;
  final String category;
  final DateTime expiryDate;

  // --- THÔNG TIN TÙY CHỌN ---
  final int? id;
  final DateTime? openedDate;
  final int? paoValue;
  final String paoUnit;        // 'months', 'days', hoặc 'hours'
  final int? quantity;
  final String? imageBase64;
  final String? owner;         // Chủ sở hữu (chọn từ danh sách & thống kê)
  final String? location;      // MỚI: Vị trí cất giữ (VD: Tủ lạnh, Tủ thuốc)
  final String? notes;         // MỚI: Ghi chú tự do nhiều dòng

  // --- CẤU HÌNH CẢNH BÁO RIÊNG (ĐÃ NÂNG CẤP) ---
  final List<AlertMilestone> alerts;

  HouseholdItem({
    this.id,
    required this.name,
    required this.category,
    required this.expiryDate,
    this.openedDate,
    this.paoValue,
    this.paoUnit = 'months',
    this.quantity,
    this.imageBase64,
    this.owner,
    this.location,
    this.notes,
    this.alerts = const [
      AlertMilestone(value: 30, unit: 'days'),
      AlertMilestone(value: 7, unit: 'days'),
    ],
  });

  // Giữ lại getter tương thích cho phần tô màu thẻ theo số ngày
  List<int> get alertDays => alerts
      .where((a) => a.unit == 'days')
      .map((a) => a.value)
      .toList();

  // Tạo bản sao vật phẩm (rất tiện khi cập nhật lại ID hoặc mã sự kiện Lịch)
  HouseholdItem copyWith({
    int? id,
    String? name,
    String? category,
    DateTime? expiryDate,
    DateTime? openedDate,
    int? paoValue,
    String? paoUnit,
    int? quantity,
    String? imageBase64,
    String? owner,
    String? location,
    String? notes,
    List<AlertMilestone>? alerts,
  }) {
    return HouseholdItem(
      id: id ?? this.id,
      name: name ?? this.name,
      category: category ?? this.category,
      expiryDate: expiryDate ?? this.expiryDate,
      openedDate: openedDate ?? this.openedDate,
      paoValue: paoValue ?? this.paoValue,
      paoUnit: paoUnit ?? this.paoUnit,
      quantity: quantity ?? this.quantity,
      imageBase64: imageBase64 ?? this.imageBase64,
      owner: owner ?? this.owner,
      location: location ?? this.location,
      notes: notes ?? this.notes,
      alerts: alerts ?? this.alerts,
    );
  }

  // Bộ nhớ đệm giúp chỉ giải mã ảnh Base64 đúng 1 lần cho mỗi vật phẩm
  Uint8List? _cachedImageBytes;
  bool _hasDecodedImage = false;

  Uint8List? get imageBytes {
    if (!_hasDecodedImage) {
      _hasDecodedImage = true;
      if (imageBase64 != null && imageBase64!.isNotEmpty) {
        try {
          _cachedImageBytes = base64Decode(imageBase64!);
        } catch (_) {
          _cachedImageBytes = null;
        }
      }
    }
    return _cachedImageBytes;
  }

  bool get hasPao => openedDate != null && paoValue != null && paoValue! > 0;
  bool get isHourlyPao => hasPao && paoUnit == 'hours';

  String get paoShortLabel {
    if (!hasPao) return '';
    switch (paoUnit) {
      case 'hours':
        return 'PAO ${paoValue}H';
      case 'days':
        return 'PAO ${paoValue}D';
      case 'months':
      default:
        return 'PAO ${paoValue}M';
    }
  }

  // Tính hạn sử dụng thực tế (Tháng / Ngày / Giờ)
  DateTime get effectiveExpiryDate {
    if (hasPao) {
      final DateTime paoExpiry;
      switch (paoUnit) {
        case 'hours':
          paoExpiry = openedDate!.add(Duration(hours: paoValue!));
          break;
        case 'days':
          paoExpiry = openedDate!.add(Duration(days: paoValue!));
          break;
        case 'months':
        default:
          paoExpiry = DateTime(
            openedDate!.year,
            openedDate!.month + paoValue!,
            openedDate!.day,
            openedDate!.hour,
            openedDate!.minute,
          );
          break;
      }

      final endOfExpiryDay = DateTime(
        expiryDate.year,
        expiryDate.month,
        expiryDate.day,
        23,
        59,
      );

      if (paoExpiry.isBefore(endOfExpiryDay)) {
        return paoExpiry;
      }
      return isHourlyPao ? endOfExpiryDay : expiryDate;
    }
    return expiryDate;
  }

  // Tính chính xác thời điểm nhắc nhở (dùng chung cho cả Thông báo app & Lịch điện thoại)
  DateTime calculateAlertTime(AlertMilestone alert) {
    final expiry = effectiveExpiryDate;
    if (alert.unit == 'hours') {
      return expiry.subtract(Duration(hours: alert.value));
    }
    // Với mốc theo ngày -> nhắc vào lúc 9:00 sáng của ngày đó
    final alertDay = expiry.subtract(Duration(days: alert.value));
    return DateTime(alertDay.year, alertDay.month, alertDay.day, 9, 0);
  }

  Duration get timeRemaining => effectiveExpiryDate.difference(DateTime.now());

  int get daysRemaining {
    final now = DateTime.now();
    final today = DateTime(now.year, now.month, now.day);
    final target = DateTime(
      effectiveExpiryDate.year,
      effectiveExpiryDate.month,
      effectiveExpiryDate.day,
    );
    return target.difference(today).inDays;
  }

  // Đóng gói dữ liệu để lưu vào SQLite hoặc xuất JSON sao lưu
  Map<String, dynamic> toMap() {
    return {
      'id': id,
      'name': name,
      'category': category,
      'expiryDate': expiryDate.toIso8601String(),
      'openedDate': openedDate?.toIso8601String(),
      'paoValue': paoValue,   // Đổi tên cột chuẩn theo ý nghĩa mới
      'paoUnit': paoUnit,
      'quantity': quantity,
      'imageBase64': imageBase64,
      'owner': owner,
      'location': location,
      'notes': notes,
      // Lưu trực tiếp danh sách mốc cảnh báo dưới dạng chuỗi JSON
      'alerts': jsonEncode(alerts.map((a) => a.toMap()).toList()),
    };
  }

  // Mở gói dữ liệu an toàn: 1 dòng lỗi định dạng không làm văng toàn bộ danh sách
  factory HouseholdItem.fromMap(Map<String, dynamic> map) {
    final String rawAlerts = (map['alerts'] ?? '').toString().trim();
    List<AlertMilestone> parsedAlerts = [];
    if (rawAlerts.isNotEmpty) {
      try {
        parsedAlerts = (jsonDecode(rawAlerts) as List<dynamic>)
            .map((e) => AlertMilestone.fromMap(Map<String, dynamic>.from(e)))
            .toList();
      } catch (_) {
        parsedAlerts = [];
      }
    }

    return HouseholdItem(
      id: (map['id'] as num?)?.toInt(),
      name: (map['name'] ?? '').toString(),
      category: (map['category'] ?? 'Other').toString(),
      expiryDate:
          DateTime.tryParse((map['expiryDate'] ?? '').toString()) ??
          DateTime.now(),
      openedDate: map['openedDate'] != null
          ? DateTime.tryParse(map['openedDate'].toString())
          : null,
      paoValue: (map['paoValue'] as num?)?.toInt(),
      paoUnit: (map['paoUnit'] ?? 'months').toString(),
      quantity: (map['quantity'] as num?)?.toInt(),
      imageBase64: map['imageBase64'] as String?,
      owner: map['owner'] as String?,
      location: map['location'] as String?,
      notes: map['notes'] as String?,
      alerts: parsedAlerts,
    );
  }
}

  List<String> _buildOwnerFilterOptions(List<HouseholdItem> items) {
    final owners = items
        .map((e) => e.owner?.trim() ?? '')
        .where((o) => o.isNotEmpty)
        .toSet()
        .toList()
      ..sort();

    final hasUnassigned = items.any(
      (e) => e.owner == null || e.owner!.trim().isEmpty,
    );

    return [
      AppConstants.filterAll,
      ...owners,
      if (hasUnassigned && owners.isNotEmpty) AppConstants.ownerUnassigned,
    ];
  }

  List<String> _buildLocationFilterOptions(List<HouseholdItem> items) {
    final locations = items
        .map((e) => e.location?.trim() ?? '')
        .where((l) => l.isNotEmpty)
        .toSet()
        .toList()
      ..sort();

    final hasUnassigned = items.any(
      (e) => e.location == null || e.location!.trim().isEmpty,
    );

    return [
      AppConstants.filterAll,
      ...locations,
      if (hasUnassigned && locations.isNotEmpty) AppConstants.locationNone,
    ];
  }

  bool get _hasActiveFilters =>
      _searchQuery.isNotEmpty ||
      _selectedCategory != AppConstants.filterAll ||
      _selectedOwner != AppConstants.filterAll ||
      _selectedLocation != AppConstants.filterAll;

  void _resetAllFilters() {
    setState(() {
      _searchController.clear();
      _searchQuery = '';
      _selectedCategory = AppConstants.filterAll;
      _selectedOwner = AppConstants.filterAll;
      _selectedLocation = AppConstants.filterAll;
    });
  }

  Future<void> _loadItems() async {
    try {
      final items = await DatabaseService.instance.getAllItems();
      if (!mounted) return;

      // Chỉ tính lại bộ lọc đúng 1 lần khi dữ liệu từ SQLite thay đổi
      final ownerOptions = _buildOwnerFilterOptions(items);
      final locationOptions = _buildLocationFilterOptions(items);

      setState(() {
        _items = items;
        _ownerFilterOptions = ownerOptions;
        _locationFilterOptions = locationOptions;
        _isLoading = false;
        if (_selectedOwner != AppConstants.filterAll &&
            !_ownerFilterOptions.contains(_selectedOwner)) {
          _selectedOwner = AppConstants.filterAll;
        }
        if (_selectedLocation != AppConstants.filterAll &&
            !_locationFilterOptions.contains(_selectedLocation)) {
          _selectedLocation = AppConstants.filterAll;
        }
      });
    } catch (_) {
      if (!mounted) return;
      setState(() => _isLoading = false);
    }
  }

  List<HouseholdItem> get _filteredItems {
    final l10n = AppLocalizations.of(context)!;
    final keyword = _searchQuery.trim().toLowerCase();

    final filtered = _items.where((item) {
      final matchesCategory = _selectedCategory == AppConstants.filterAll ||
          item.category == _selectedCategory;

      final ownerName = item.owner?.trim() ?? '';
      final bool matchesOwner;
      if (_selectedOwner == AppConstants.filterAll) {
        matchesOwner = true;
      } else if (_selectedOwner == AppConstants.ownerUnassigned) {
        matchesOwner = ownerName.isEmpty;
      } else {
        matchesOwner = ownerName == _selectedOwner;
      }

      final locationName = item.location?.trim() ?? '';
      final bool matchesLocation;
      if (_selectedLocation == AppConstants.filterAll) {
        matchesLocation = true;
      } else if (_selectedLocation == AppConstants.locationNone) {
        matchesLocation = locationName.isEmpty;
      } else {
        matchesLocation = locationName == _selectedLocation;
      }

      final bool matchesSearch;
      if (keyword.isEmpty) {
        matchesSearch = true;
      } else {
        final searchableText = [
          item.name,
          item.category,
          localizeCategory(l10n, item.category),
          item.owner ?? '',
          if (item.owner != null && item.owner!.isNotEmpty)
            localizeOwner(l10n, item.owner!),
          item.location ?? '',
          if (item.location != null && item.location!.isNotEmpty)
            localizeLocation(l10n, item.location!),
          item.notes ?? '',
        ].join(' ').toLowerCase();
        matchesSearch = searchableText.contains(keyword);
      }

      return matchesCategory && matchesOwner && matchesLocation && matchesSearch;
    }).toList();

    // So sánh trực tiếp mốc ngày hết hạn thay vì gọi DateTime.now() nhiều lần qua timeRemaining
    filtered.sort(
      (a, b) => a.effectiveExpiryDate.compareTo(b.effectiveExpiryDate),
    );
    return filtered;
  }

  Future<void> _openAddEditScreen({HouseholdItem? existingItem}) async {
    final result = await Navigator.push(
      context,
      MaterialPageRoute(
        builder: (context) => AddEditScreen(existingItem: existingItem),
      ),
    );

    if (result == null) return;

    if (result == 'delete' && existingItem?.id != null) {
      await DatabaseService.instance.deleteItem(existingItem!.id!);
      await NotifyService.instance.cancelItemAlerts(existingItem.id!);
      await CalendarService.instance.deleteItemCalendarEvents(existingItem);
      await LogService.instance.addLog(
        'DELETE',
        encodeLogMessage('logItemDeleted', {'name': existingItem.name}),
      );
    } else if (result is HouseholdItem) {
      if (existingItem == null) {
        final newId = await DatabaseService.instance.insertItem(result);
        var savedItem = result.copyWith(id: newId);

        final syncResult = await CalendarService.instance
            .syncItemCalendarEvents(savedItem);
        savedItem = savedItem.copyWith(alerts: syncResult.alerts);
        await DatabaseService.instance.updateItem(savedItem);

        await NotifyService.instance.scheduleItemAlerts(savedItem);
        await LogService.instance.addLog(
          'ADD',
          encodeLogMessage('logItemAdded', {
            'name': savedItem.name,
            'category': savedItem.category,
          }),
        );

        if (syncResult.hasError && mounted) {
          _showCalendarError(syncResult.errorKey!, savedItem.name);
        }
      } else {
        final syncResult = await CalendarService.instance
            .syncItemCalendarEvents(
              result,
              previousAlerts: existingItem.alerts,
            );
        final updatedItem = result.copyWith(alerts: syncResult.alerts);

        await DatabaseService.instance.updateItem(updatedItem);
        await NotifyService.instance.scheduleItemAlerts(updatedItem);
        await LogService.instance.addLog(
          'UPDATE',
          encodeLogMessage('logItemUpdated', {'name': updatedItem.name}),
        );

        if (syncResult.hasError && mounted) {
          _showCalendarError(syncResult.errorKey!, updatedItem.name);
        }
      }
    }

    await _loadItems();
  }

class BackupService {
  static final BackupService instance = BackupService._init();
  BackupService._init();

  Future<int> exportBackup() async {
    final items = await DatabaseService.instance.getAllItems();

    final Map<String, dynamic> backupPayload = {
      'app': 'MinExpiryTracker',
      'version': 1,
      'exportedAt': DateTime.now().toIso8601String(),
      'items': items.map((item) => item.toMap()).toList(),
    };

    final String jsonString = jsonEncode(backupPayload);
    final String dateStamp = DateFormat('yyyy-MM-dd').format(DateTime.now());
    final String fileName = 'MinExpiryTracker_$dateStamp.json';

    final Directory tempDir = await getTemporaryDirectory();
    final File backupFile = File('${tempDir.path}/$fileName');
    await backupFile.writeAsString(jsonString);

    await SharePlus.instance.share(
      ShareParams(
        files: [XFile(backupFile.path)],
        subject: 'MinExpiryTracker Backup ($dateStamp)',
      ),
    );

    await LogService.instance.addLog(
      'BACKUP',
      encodeLogMessage('logBackupExported', {
        'count': items.length,
        'fileName': fileName,
      }),
    );

    return items.length;
  }

  Future<int?> importBackup() async {
    final FilePickerResult? result = await FilePicker.pickFiles(
      type: FileType.any,
    );

    if (result == null || result.files.single.path == null) {
      return null;
    }

    final File file = File(result.files.single.path!);
    final String content = await file.readAsString();

    final dynamic decoded = jsonDecode(content);
    if (decoded is! Map<String, dynamic> ||
        decoded['app'] != 'MinExpiryTracker' ||
        decoded['items'] is! List) {
      await LogService.instance.addLog(
        'ERROR',
        encodeLogMessage('logRestoreInvalid'),
      );
      throw const FormatException('Invalid MinExpiryTracker backup file.');
    }

    final List<dynamic> rawItems = decoded['items'];
    final List<HouseholdItem> restoredItems = rawItems
        .map((e) => HouseholdItem.fromMap(Map<String, dynamic>.from(e)))
        .toList();

    // Bước 1: Dọn các sự kiện Lịch của dữ liệu cũ đang nằm trên máy hiện tại
    final existingItems = await DatabaseService.instance.getAllItems();
    for (final oldItem in existingItems) {
      await CalendarService.instance.deleteItemCalendarEvents(oldItem);
    }

    // Bước 2: Xóa sạch mã Lịch cũ (calendarEventId & calendarId) từ máy trước trong file backup
    final cleanedItems = restoredItems.map((item) {
      final resetAlerts = item.alerts
          .map((a) => a.copyWith(clearCalendarEventId: true))
          .toList();
      return item.copyWith(alerts: resetAlerts);
    }).toList();

    // Bước 3: Ghi đè toàn bộ vào SQLite trong 1 giao dịch nguyên tử
    final savedItems =
        await DatabaseService.instance.replaceAllItems(cleanedItems);

    // Bước 4: Đặt lại toàn bộ Thông báo offline và đồng bộ mới sang Lịch của máy hiện tại
    await NotifyService.instance.cancelAllAlerts();
    bool calendarUnavailable = false;

    for (final item in savedItems) {
      // Truyền cancelExisting: false vì vừa gọi cancelAllAlerts() ngay phía trên -> khôi phục nhanh hơn
      await NotifyService.instance.scheduleItemAlerts(
        item,
        cancelExisting: false,
      );

      if (!calendarUnavailable && item.alerts.any((a) => a.addToCalendar)) {
        final syncResult =
            await CalendarService.instance.syncItemCalendarEvents(item);

        if (syncResult.errorKey == 'logCalendarDenied' ||
            syncResult.errorKey == 'logCalendarNoWritable') {
          calendarUnavailable = true;
        } else {
          final syncedItem = item.copyWith(alerts: syncResult.alerts);
          await DatabaseService.instance.updateItem(syncedItem);
        }
      }
    }

    await LogService.instance.addLog(
      'BACKUP',
      encodeLogMessage('logBackupRestored', {'count': savedItems.length}),
    );

    // Thu hồi dung lượng trống của dữ liệu ảnh cũ vừa bị ghi đè
    await DatabaseService.instance.compactDatabase();

    return savedItems.length;
  }
}

import 'locale_service.dart';

// Gói kết quả đồng bộ Lịch để báo ngược trạng thái lỗi ra màn hình chính
class CalendarSyncResult {
  final List<AlertMilestone> alerts;
  final String? errorKey; // null nếu thành công hoặc chỉ đang bỏ tick Lịch

  const CalendarSyncResult({
    required this.alerts,
    this.errorKey,
  });

  bool get hasError => errorKey != null;
}

class _CalendarLookupResult {
  final String? calendarId;
  final String? errorKey;

  const _CalendarLookupResult({this.calendarId, this.errorKey});
}

class CalendarService {
  static final CalendarService instance = CalendarService._init();
  CalendarService._init();

  final DeviceCalendarPlugin _plugin = DeviceCalendarPlugin();

  // Nhận diện sổ Lịch thuộc Google Calendar
  bool _isGoogleCalendar(Calendar c) {
    final type = (c.accountType ?? '').toLowerCase();
    final accName = (c.accountName ?? '').toLowerCase();
    return type.contains('google') || accName.contains('@gmail.com');
  }

  // Nhận diện Lịch mặc định của Samsung hoặc Lịch cục bộ trên điện thoại
  bool _isDeviceOrSamsungCalendar(Calendar c) {
    final type = (c.accountType ?? '').toLowerCase();
    final accName = (c.accountName ?? '').toLowerCase();
    final calName = (c.name ?? '').toLowerCase();

    return type.contains('local') ||
        type.contains('samsung') ||
        type.contains('osp') ||
        type.contains('sec') ||
        type.contains('phone') ||
        accName.contains('local') ||
        accName.contains('phone') ||
        accName.contains('samsung') ||
        calName.contains('my calendar') ||
        calName.contains('lịch của tôi') ||
        calName.contains('samsung');
  }

  Calendar _pickDefaultOrFirst(List<Calendar> list) {
    return list.firstWhere(
      (c) => c.isDefault == true,
      orElse: () => list.first,
    );
  }

  // Chọn sổ Lịch ưu tiên: Lịch máy/Samsung -> Google Calendar -> Lịch khả dụng khác
  Calendar _pickPreferredCalendar(List<Calendar> writable) {
    final deviceCalendars = writable.where(_isDeviceOrSamsungCalendar).toList();
    if (deviceCalendars.isNotEmpty) {
      return _pickDefaultOrFirst(deviceCalendars);
    }

    final defaultNonGoogle = writable
        .where((c) => c.isDefault == true && !_isGoogleCalendar(c))
        .toList();
    if (defaultNonGoogle.isNotEmpty) {
      return defaultNonGoogle.first;
    }

    final googleCalendars = writable.where(_isGoogleCalendar).toList();
    if (googleCalendars.isNotEmpty) {
      return _pickDefaultOrFirst(googleCalendars);
    }

    return _pickDefaultOrFirst(writable);
  }

  Future<String?> _createLocalFallbackCalendar() async {
    try {
      final result = await _plugin.createCalendar(
        'MinExpiryTracker',
        localAccountName: 'MinExpiryTracker',
      );
      if (result.isSuccess && result.data != null && result.data!.isNotEmpty) {
        return result.data;
      }
    } catch (_) {}
    return null;
  }

  // Tìm ID của Lịch khả dụng (thêm cờ logFailure để không ghi log lỗi khi chỉ đang dọn sự kiện cũ)
  Future<_CalendarLookupResult> _getWritableCalendarId({
    bool allowCreateFallback = true,
    bool logFailure = true,
  }) async {
    try {
      var permissionResult = await _plugin.hasPermissions();
      if (permissionResult.isSuccess && permissionResult.data != true) {
        // Nếu không phải thao tác tạo mới (logFailure == false) thì không bật hộp thoại xin quyền
        if (!logFailure) {
          return const _CalendarLookupResult();
        }
        permissionResult = await _plugin.requestPermissions();
        if (!permissionResult.isSuccess || permissionResult.data != true) {
          await LogService.instance.addLog(
            'CALENDAR',
            encodeLogMessage('logCalendarDenied'),
          );
          return const _CalendarLookupResult(errorKey: 'logCalendarDenied');
        }
      }

      final calendarsResult = await _plugin.retrieveCalendars();
      final List<Calendar> calendars = calendarsResult.data ?? <Calendar>[];

      final writable = calendars
          .where((c) => c.id != null && c.id!.isNotEmpty && c.isReadOnly != true)
          .toList();

      if (writable.isNotEmpty) {
        final selected = _pickPreferredCalendar(writable);
        return _CalendarLookupResult(calendarId: selected.id);
      }

      if (allowCreateFallback) {
        final localId = await _createLocalFallbackCalendar();
        if (localId != null) {
          return _CalendarLookupResult(calendarId: localId);
        }
      }
    } catch (_) {}

    if (logFailure) {
      await LogService.instance.addLog(
        'CALENDAR',
        encodeLogMessage('logCalendarNoWritable'),
      );
      return const _CalendarLookupResult(errorKey: 'logCalendarNoWritable');
    }
    return const _CalendarLookupResult();
  }

  // Hàm dọn sự kiện cũ an toàn tuyệt đối (không bao giờ ném lỗi ra giao diện)
  Future<int> _deleteSavedEvents(
    List<AlertMilestone> alerts, {
    String? fallbackCalendarId,
  }) async {
    final alertsWithIds = alerts
        .where((a) => a.calendarEventId != null && a.calendarEventId!.isNotEmpty)
        .toList();
    if (alertsWithIds.isEmpty) return 0;

    try {
      final perm = await _plugin.hasPermissions();
      if (!perm.isSuccess || perm.data != true) return 0;
    } catch (_) {
      return 0;
    }

    String? sharedFallbackId = fallbackCalendarId;
    int removedCount = 0;

    for (final alert in alertsWithIds) {
      String? targetCalId = alert.calendarId;
      if (targetCalId == null || targetCalId.isEmpty) {
        sharedFallbackId ??= (await _getWritableCalendarId(
          allowCreateFallback: false,
          logFailure: false,
        ))
            .calendarId;
        targetCalId = sharedFallbackId;
      }

      if (targetCalId == null || targetCalId.isEmpty) continue;

      try {
        final res = await _plugin.deleteEvent(
          targetCalId,
          alert.calendarEventId!,
        );
        if (res.isSuccess || res.data == true) {
          removedCount++;
        }
      } catch (_) {
        // Bỏ qua êm ái nếu sự kiện đã bị người dùng xóa thủ công trên app Lịch
      }
    }

    return removedCount;
  }

  // Đồng bộ tự động các mốc được tick "Thêm vào Lịch" của 1 vật phẩm
  Future<CalendarSyncResult> syncItemCalendarEvents(
    HouseholdItem item, {
    List<AlertMilestone>? previousAlerts,
  }) async {
    final hasOldEvents = previousAlerts != null &&
        previousAlerts.any(
          (a) => a.calendarEventId != null && a.calendarEventId!.isNotEmpty,
        );
    final hasNewRequests = item.alerts.any((a) => a.addToCalendar);

    // Trường hợp 1: Không có lịch cũ và cũng không tick mới -> thoát êm
    if (!hasOldEvents && !hasNewRequests) {
      return CalendarSyncResult(alerts: item.alerts);
    }

    // Trường hợp 2 (ĐÃ SỬA LOGIC): Người dùng BỎ DẤU TICK tất cả mốc Lịch
    // -> Cố gắng dọn sự kiện cũ trong âm thầm, TUYỆT ĐỐI không báo lỗi lên màn hình
    if (!hasNewRequests) {
      if (previousAlerts != null) {
        final removedCount = await _deleteSavedEvents(previousAlerts);
        if (removedCount > 0) {
          await LogService.instance.addLog(
            'CALENDAR',
            encodeLogMessage('logCalendarRemoved', {
              'count': removedCount,
              'name': item.name,
            }),
          );
        }
      }

      return CalendarSyncResult(
        alerts: item.alerts
            .map((a) => a.copyWith(clearCalendarEventId: true))
            .toList(),
        errorKey: null,
      );
    }

    // Trường hợp 3: Người dùng CÓ tick ít nhất 1 mốc muốn ghi lên Lịch
    final lookup = await _getWritableCalendarId(
      allowCreateFallback: true,
      logFailure: true,
    );

    if (lookup.calendarId == null) {
      return CalendarSyncResult(
        alerts: item.alerts
            .map((a) => a.copyWith(clearCalendarEventId: true))
            .toList(),
        errorKey: lookup.errorKey ?? 'logCalendarNoWritable',
      );
    }

    final calendarId = lookup.calendarId!;

    // Xóa các sự kiện cũ của lần lưu trước (nếu có) trước khi tạo bộ sự kiện mới
    if (previousAlerts != null) {
      await _deleteSavedEvents(previousAlerts, fallbackCalendarId: calendarId);
    }

    final List<AlertMilestone> updatedAlerts = [];
    final now = DateTime.now();
    // Lấy từ điển & định dạng ngày theo ngôn ngữ hiện tại (VN: dd/MM/yyyy, EN: MMM dd, yyyy)
    final l10n = LocaleService.instance.currentL10n;
    final isVi = LocaleService.instance.effectiveLocale.languageCode == 'vi';
    final dateFmt = DateFormat(isVi ? 'dd/MM/yyyy HH:mm' : 'MMM dd, yyyy HH:mm');
    final categoryLabel = localizeCategory(l10n, item.category);
    String? eventErrorKey;

    for (final alert in item.alerts) {
      if (!alert.addToCalendar) {
        updatedAlerts.add(alert.copyWith(clearCalendarEventId: true));
        continue;
      }

      final alertTime = item.calculateAlertTime(alert);
      if (!alertTime.isAfter(now)) {
        await LogService.instance.addLog(
          'CALENDAR',
          encodeLogMessage('logCalendarSkippedPast', {
            'alertValue': alert.value,
            'alertUnit': alert.unit,
            'name': item.name,
          }),
        );
        updatedAlerts.add(alert.copyWith(clearCalendarEventId: true));
        continue;
      }

      final tzStart = tz.TZDateTime.from(alertTime, tz.local);
      final tzEnd = tzStart.add(const Duration(minutes: 30));
      final alertLabel = localizeAlertLabel(l10n, alert);

      final details = <String>[
        l10n.calItemLine(item.name, categoryLabel),
        l10n.calExpiresLine(dateFmt.format(item.effectiveExpiryDate)),
        l10n.calReminderLine(alertLabel),
        if (item.owner != null && item.owner!.isNotEmpty)
          l10n.calOwnerLine(localizeOwner(l10n, item.owner!)),
        if (item.location != null && item.location!.isNotEmpty)
          l10n.calLocationLine(localizeLocation(l10n, item.location!)),
        if (item.notes != null && item.notes!.isNotEmpty)
          l10n.calNotesLine(item.notes!),
      ].join('\n');

      final event = Event(
        calendarId,
        title: '[MinExpiry] ${item.name} ($alertLabel)',
        description: details,
        start: tzStart,
        end: tzEnd,
        reminders: [Reminder(minutes: 0)],
      );

      try {
        final result = await _plugin.createOrUpdateEvent(event);

        // ĐÃ SỬA: Coi là thành công nếu plugin báo isSuccess HOẶC đã trả về mã sự kiện hợp lệ
        final String? createdEventId =
            (result?.data != null && result!.data!.trim().isNotEmpty)
                ? result.data!.trim()
                : null;
        final bool isCreated =
            (result?.isSuccess == true) || createdEventId != null;

        if (isCreated) {
          updatedAlerts.add(
            alert.copyWith(
              calendarEventId: createdEventId,
              calendarId: createdEventId != null ? calendarId : null,
            ),
          );
          await LogService.instance.addLog(
            'CALENDAR',
            encodeLogMessage('logCalendarAdded', {
              'alertValue': alert.value,
              'alertUnit': alert.unit,
              'name': item.name,
            }),
          );
        } else {
          eventErrorKey = 'logCalendarFailed';
          updatedAlerts.add(alert.copyWith(clearCalendarEventId: true));
          await LogService.instance.addLog(
            'ERROR',
            encodeLogMessage('logCalendarFailed', {'name': item.name}),
          );
        }
      } catch (_) {
        eventErrorKey = 'logCalendarFailed';
        updatedAlerts.add(alert.copyWith(clearCalendarEventId: true));
        await LogService.instance.addLog(
          'ERROR',
          encodeLogMessage('logCalendarFailed', {'name': item.name}),
        );
      }
    }

    return CalendarSyncResult(
      alerts: updatedAlerts,
      errorKey: eventErrorKey,
    );
  }

  // Xóa toàn bộ sự kiện Lịch gắn với 1 vật phẩm (dùng khi xóa món đồ)

import 'package:path/path.dart';

class DatabaseService {
  static final DatabaseService instance = DatabaseService._init();
  static Database? _database;

  DatabaseService._init();

  Future<Database> get database async {
    if (_database != null) return _database!;
    _database = await _initDB('min_expiry_tracker.db');
    return _database!;
  }

  Future<Database> _initDB(String fileName) async {
    final dbPath = await getDatabasesPath();
    final path = join(dbPath, fileName);

    return await openDatabase(
      path,
      version: 1, // Giữ version 1 sạch sẽ trong giai đoạn đang phát triển
      onCreate: _createDB,
    );
  }

  // Tạo mới 2 bảng hoàn chỉnh ngay từ đầu: items và app_logs
  Future<void> _createDB(Database db, int version) async {
    await db.execute('''
      CREATE TABLE items (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        name TEXT NOT NULL,
        category TEXT NOT NULL,
        expiryDate TEXT NOT NULL,
        openedDate TEXT,
        paoValue INTEGER,
        paoUnit TEXT DEFAULT 'months',
        quantity INTEGER,
        imageBase64 TEXT,
        owner TEXT,
        location TEXT,
        notes TEXT,
        alerts TEXT
      )
    ''');

    await db.execute('''
      CREATE TABLE app_logs (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        timestamp TEXT NOT NULL,
        action TEXT NOT NULL,
        message TEXT NOT NULL
      )
    ''');
  }

  // 1. CREATE
  Future<int> insertItem(HouseholdItem item) async {
    final db = await instance.database;
    final data = item.toMap()..remove('id');
    return await db.insert('items', data);
  }

  // 2. READ
  Future<List<HouseholdItem>> getAllItems() async {
    final db = await instance.database;
    final maps = await db.query('items');
    return maps.map((map) => HouseholdItem.fromMap(map)).toList();
  }

  // 3. UPDATE
  Future<int> updateItem(HouseholdItem item) async {
    final db = await instance.database;
    return await db.update(
      'items',
      item.toMap(),
      where: 'id = ?',
      whereArgs: [item.id],
    );
  }

  // 4. DELETE
  Future<int> deleteItem(int id) async {
    final db = await instance.database;
    return await db.delete(
      'items',
      where: 'id = ?',
      whereArgs: [id],
    );
  }

  // 5. Xóa toàn bộ danh sách vật phẩm (dùng trước khi Restore)
  Future<void> clearAllItems() async {
    final db = await instance.database;
    await db.delete('items');
  }

  // 6. Lấy danh sách Chủ sở hữu đã dùng (không trùng lặp) để chọn nhanh & lọc
  Future<List<String>> getDistinctOwners() async {
    final db = await instance.database;
    final rows = await db.rawQuery(
      "SELECT DISTINCT owner FROM items WHERE owner IS NOT NULL AND TRIM(owner) != '' ORDER BY owner ASC",
    );
    return rows.map((r) => (r['owner'] as String).trim()).toList();
  }

  // 7. Lấy danh sách Vị trí đã dùng (không trùng lặp) để chọn nhanh
  Future<List<String>> getDistinctLocations() async {
    final db = await instance.database;
    final rows = await db.rawQuery(
      "SELECT DISTINCT location FROM items WHERE location IS NOT NULL AND TRIM(location) != '' ORDER BY location ASC",
    );
    return rows.map((r) => (r['location'] as String).trim()).toList();
  }

  // 8. Thay thế toàn bộ vật phẩm trong 1 giao dịch nguyên tử (dùng khi Restore)
  Future<List<HouseholdItem>> replaceAllItems(List<HouseholdItem> items) async {
    final db = await instance.database;
    return await db.transaction((txn) async {
      await txn.delete('items');
      final List<HouseholdItem> savedItems = [];
      for (final item in items) {
        final data = item.toMap()..remove('id');
        final newId = await txn.insert('items', data);
        savedItems.add(item.copyWith(id: newId));
      }
      return savedItems;
    });
  }

  // 9. Thu hồi dung lượng trống trong file SQLite (giúp file .db co gọn thực tế trên bộ nhớ máy)
  Future<void> compactDatabase() async {
    try {
      final db = await instance.database;
      await db.execute('VACUUM');
    } catch (_) {
      // Lệnh VACUUM không chạy được khi đang mở transaction -> bỏ qua êm ái nếu bận
    }
  }
}

  Locale get effectiveLocale {
    final code = localeNotifier.value?.languageCode ??
        WidgetsBinding.instance.platformDispatcher.locale.languageCode;
    return code == 'vi' ? const Locale('vi') : const Locale('en');
  }

  // MỚI: Lấy thẳng bộ từ điển mà không cần BuildContext (dùng cho NotifyService & CalendarService)

    final dir = await getApplicationDocumentsDirectory();
    return File('${dir.path}/app_locale.txt');
  }

  Future<void> loadSavedLocale() async {
    try {
      final file = await _getPrefFile();
      if (!await file.exists()) return;

      final code = (await file.readAsString()).trim();
      if (code == 'en' || code == 'vi') {
        localeNotifier.value = Locale(code);
      } else {
        localeNotifier.value = null;
      }
    } catch (_) {
      localeNotifier.value = null;
    }
  }

    localeNotifier.value =
        languageCode == null ? null : Locale(languageCode);

    try {
      final file = await _getPrefFile();
      await file.writeAsString(languageCode ?? '');
    } catch (_) {}
  }

      message: map['message'],
    );

class LogService {
  static final LogService instance = LogService._init();
  LogService._init();

  // Giới hạn tối đa 200 bản ghi trong SQLite để file database không phình to theo năm tháng
  static const int _maxStoredLogs = 200;

  Future<void> addLog(String action, String message) async {
    try {
      final db = await DatabaseService.instance.database;
      await db.insert('app_logs', {
        'timestamp': DateTime.now().toIso8601String(),
        'action': action,
        'message': message,
      });

      // Tự động xóa các dòng log cũ vượt quá giới hạn 200 dòng mới nhất
      await db.rawDelete('''
        DELETE FROM app_logs
        WHERE id NOT IN (
          SELECT id FROM app_logs ORDER BY id DESC LIMIT ?
        )
      ''', [_maxStoredLogs]);
    } catch (_) {
      // Đảm bảo lỗi ghi log không bao giờ làm gián đoạn chức năng chính
    }
  }

    final db = await DatabaseService.instance.database;
    final rows = await db.query(
      'app_logs',
      orderBy: 'id DESC',
      limit: limit,
    );
    return rows.map((row) => AppLog.fromMap(row)).toList();
  }

  Future<void> clearLogs() async {
    final db = await DatabaseService.instance.database;
    await db.delete('app_logs');
    await DatabaseService.instance.compactDatabase(); // Thu hồi dung lượng trống sau khi xóa sạch Log
  }

class NotifyService {
  static final NotifyService instance = NotifyService._init();
  NotifyService._init();

  final FlutterLocalNotificationsPlugin _plugin =
      FlutterLocalNotificationsPlugin();

  static const AndroidNotificationDetails _androidDetails =
      AndroidNotificationDetails(
    'expiry_alerts_channel',
    'Expiry Reminders',
    channelDescription: 'Notifications before your household items expire',
    importance: Importance.high,
    priority: Priority.high,
  );

  static const NotificationDetails _notificationDetails = NotificationDetails(
    android: _androidDetails,
  );

  Future<void> init() async {
    tz_data.initializeTimeZones();

    const androidInit = AndroidInitializationSettings('@mipmap/ic_launcher');
    const initSettings = InitializationSettings(android: androidInit);

    await _plugin.initialize(settings: initSettings);

    await _plugin
        .resolvePlatformSpecificImplementation<
            AndroidFlutterLocalNotificationsPlugin>()
        ?.requestNotificationsPermission();
  }

  Future<void> showTestNotification() async {
    final l10n = LocaleService.instance.currentL10n;

    await _plugin.show(
      id: 0,
      title: l10n.testNotifyTitle,
      body: l10n.testNotifyBody,
      notificationDetails: _notificationDetails,
    );
    await LogService.instance.addLog(
      'NOTIFY',
      encodeLogMessage('logTestNotifySent'),
    );
  }

  Future<void> cancelItemAlerts(int itemId) async {
    final pending = await _plugin.pendingNotificationRequests();
    for (final request in pending) {
      if (request.id ~/ 100 == itemId) {
        await _plugin.cancel(id: request.id);
      }
    }
  }

  Future<void> scheduleItemAlerts(
    HouseholdItem item, {
    int? overrideId,
    bool cancelExisting = true,
    bool logAction = true,
  }) async {
    final int? itemId = overrideId ?? item.id;
    if (itemId == null) return;

    // Khi vừa chạy cancelAllAlerts() trước đó -> truyền cancelExisting: false để đỡ truy vấn lặp
    if (cancelExisting) {
      await cancelItemAlerts(itemId);
    }

    final l10n = LocaleService.instance.currentL10n;
    final categoryLabel = localizeCategory(l10n, item.category);
    final now = DateTime.now();
    int scheduledCount = 0;

    for (int i = 0; i < item.alerts.length; i++) {
      final alert = item.alerts[i];
      final scheduledLocalTime = item.calculateAlertTime(alert);

      if (!scheduledLocalTime.isAfter(now)) continue;

      final tzScheduledDate = tz.TZDateTime.from(scheduledLocalTime, tz.local);
      final notificationId = itemId * 100 + i;

      final String bodyText;
      if (alert.unit == 'hours') {
        bodyText = alert.value == 1
            ? l10n.expiresInOneHour(item.name)
            : l10n.expiresInHours(item.name, alert.value);
      } else {
        bodyText = alert.value == 1
            ? l10n.expiresTomorrow(item.name)
            : l10n.expiresInDays(item.name, alert.value);
      }

      await _plugin.zonedSchedule(
        id: notificationId,
        title: l10n.expiryAlertTitle(categoryLabel),
        body: bodyText,
        scheduledDate: tzScheduledDate,
        notificationDetails: _notificationDetails,
        androidScheduleMode: AndroidScheduleMode.inexactAllowWhileIdle,
      );
      scheduledCount++;
    }

    if (scheduledCount > 0 && logAction) {
      await LogService.instance.addLog(
        'NOTIFY',
        encodeLogMessage('logRemindersScheduled', {
          'count': scheduledCount,
          'name': item.name,
        }),
      );
    }
  }

  // MỚI: Tự đặt lại toàn bộ thông báo đang chờ sang ngôn ngữ hiện tại (dùng khi đổi ngôn ngữ)
  Future<void> rescheduleAllAlerts() async {
    try {
      final items = await DatabaseService.instance.getAllItems();
      await cancelAllAlerts();
      for (final item in items) {
        await scheduleItemAlerts(
          item,
          cancelExisting: false,
          logAction: false,
        );
      }
    } catch (_) {
      // Giữ thao tác đổi ngôn ngữ luôn mượt, không bị gián đoạn nếu có lỗi phụ
    }
  }

  Future<void> cancelAllAlerts() async {
    await _plugin.cancelAll();
  }
}

  Color _getStatusColor(bool isDark) {
    final redCritical = isDark ? Colors.red.shade300 : Colors.red.shade700;
    final redSoon = isDark ? Colors.red.shade300 : Colors.red.shade600;
    final orangeWarn = isDark ? Colors.orange.shade300 : Colors.orange.shade800;
    final greenSafe = isDark ? Colors.green.shade300 : Colors.green.shade700;

    if (item.isHourlyPao) {
      final remaining = item.timeRemaining;
      if (remaining.isNegative || remaining.inHours < 3) {
        return redCritical;
      }
      if (remaining.inHours < 12) {
        return orangeWarn;
      }
      return greenSafe;
    }

    final days = item.daysRemaining;
    if (days <= 0) return redCritical;

    if (item.alertDays.isNotEmpty) {
      final sortedAlerts = List<int>.from(item.alertDays)..sort();
      if (sortedAlerts.length >= 2 && days <= sortedAlerts.first) {
        return redSoon;

      if (days <= sortedAlerts.last) {

import 'package:flutter/material.dart';

// Widget dùng chung cho các khu vực chọn thẻ (Chủ sở hữu, Vị trí...)
class TagSelectorSection extends StatelessWidget {
  final String title;
  final List<String> options;
  final String? selectedValue;
  final String Function(String value) labelBuilder;
  final String addButtonLabel;
  final IconData addButtonIcon;
  final ValueChanged<String?> onSelected;
  final VoidCallback onAddPressed;

  const TagSelectorSection({
    super.key,
    required this.title,
    required this.options,
    required this.selectedValue,
    required this.labelBuilder,
    required this.addButtonLabel,
    required this.addButtonIcon,
    required this.onSelected,
    required this.onAddPressed,
  });

  @override
  Widget build(BuildContext context) {
    return Column(
      crossAxisAlignment: CrossAxisAlignment.start,
      children: [
        Text(
          title,
          style: const TextStyle(
            fontSize: 15,
            fontWeight: FontWeight.bold,
          ),
        ),
        const SizedBox(height: 6),
        Wrap(
          spacing: 8,
          runSpacing: 4,
          children: [
            ...options.map((item) {
              final isSelected = selectedValue == item;
              return ChoiceChip(
                label: Text(labelBuilder(item)),
                selected: isSelected,
                onSelected: (selected) {
                  onSelected(selected ? item : null);
                },
              );
            }),
            ActionChip(
              avatar: Icon(addButtonIcon, size: 18),
              label: Text(addButtonLabel),
              onPressed: onAddPressed,
            ),
          ],
        ),
      ],
    );
  }
}
